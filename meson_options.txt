option('platform',
	type : 'string',
	value : 'pc_x86_64',
	description : 'Target platform (subdir under platforms/) to configure kernel build.')

option('tests',
	type : 'boolean',
	value : true,
	description : 'Build native test targets.')

option('test_sanitizers',
	type : 'boolean',
	value : false,
	description : 'Instrument native tests with AddressSanitizer and UndefinedBehaviorSanitizer.')

option('kernel_selftests',
	type : 'boolean',
	value : false,
	description : 'Compile in-kernel selftest suites into the kernel binary.')

option('kernel_selftests_autorun',
	type : 'boolean',
	value : false,
	description : 'Append kernel.selftest=1 to the generated boot image command line.')

option('kernel_selftests_suite',
	type : 'string',
	value : '',
	description : 'Append kernel.selftest.suite=<name> to the generated boot image command line.')

option('kernel_selftests_iommu',
	type : 'string',
	value : '',
	description : 'Append kernel.selftest.iommu=<kind> to require an IOMMU backend during selftests.')

option('kernel_boot_debug',
	type : 'boolean',
	value : false,
	description : 'Append loglevel=debug to the generated boot image command line for verbose boot diagnostics.')

option('kernel_trace',
	type : 'boolean',
	value : false,
	description : 'Compile kernel tracing and enable one boot-time profiling session.')

option('kernel_trace_duration_ms',
	type : 'integer',
	value : 5000,
	min : 1,
	description : 'Duration of an enabled boot-time profiling session in milliseconds.')

option('kernel_trace_buffer_kb',
	type : 'integer',
	value : 256,
	min : 4,
	description : 'Trace buffer capacity per CPU in KiB.')

option('kernel_trace_transport',
	type : 'combo',
	choices : ['qemu', 'serial'],
	value : 'qemu',
	description : 'Completed-trace export transport.')

option('crash_server_url',
	type : 'string',
	value : '',
	description : 'Base URL used to print browser-decodable crash frame links.')
